class Solution {
    public void reverseArray(int arr[]) {
        // code here
        int l= 0;
        int o =0;
        //sequential success 5
        int r=arr.length-1;
        while(l<r){
            int temp = arr[l];
            arr[l] = arr[r];
            arr[r] = temp;
            l++;
            r--;
        }
    }
}
