class Solution {
    public boolean isPossible(int k, int[] a, int[] b) {
        //Permutatuion pair sum
        Arrays.sort(a);
        descsort(b);
        for(int i=0;i<a.length;i++){
            if(a[i]+b[i]<k){
                return false;
            }
        }
        return true;
    }
    public static void descsort(int[] a){
        Arrays.sort(a);
        int l=0;
        int r=a.length-1;
        while(l<r){
            swap(a,l,r);
            l++;
            r--;
        }
    }
    public static void swap(int[] a,int l,int r){
        int temp = a[l];
        a[l] = a[r];
        a[r] = temp;
    }
}
