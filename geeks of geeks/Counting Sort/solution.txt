class Solution {
    String countSort(String arr) {
        int[] count = new int[26];

        // Count frequency
        for (char ch : arr.toCharArray()) {
            count[ch - 'a']++;
        }

        // Build sorted string
        StringBuilder ans = new StringBuilder();

        for (int i = 0; i < 26; i++) {
            while (count[i] > 0) {
                ans.append((char) ('a' + i));
                count[i]--;
            }
        }

        return ans.toString();
    }
}
